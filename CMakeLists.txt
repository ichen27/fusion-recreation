cmake_minimum_required(VERSION 3.24)
project(FusionFoundation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
include(cmake/Dependencies.cmake)
include(cmake/PlaneGCS.cmake)
add_library(cad src/kernel.cpp src/sketch.cpp src/document.cpp src/selection.cpp src/codec.cpp src/checksum.cpp src/storage.cpp src/exchange.cpp src/commands.cpp src/protocol.cpp)
target_include_directories(cad PUBLIC include PRIVATE src ${OpenCASCADE_INCLUDE_DIR} ${CAD_DEPS}/${JSON_DIR}/single_include)
target_link_directories(cad PUBLIC ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(cad PUBLIC TKDESTEP TKXSBase TKCAF TKLCAF TKCDF planegcs TKPrim TKBool TKTopAlgo TKBRep TKG3d TKMath TKernel)
function(cad_test name)
 add_executable(${name}_test tests/${name}_test.cpp)
 target_link_libraries(${name}_test PRIVATE cad)
 target_include_directories(${name}_test PRIVATE src ${OpenCASCADE_INCLUDE_DIR} ${CAD_DEPS}/${JSON_DIR}/single_include)
 add_test(NAME ${name} COMMAND ${name}_test)
endfunction()
cad_test(kernel)
cad_test(sketch)
cad_test(document)
cad_test(selection)
cad_test(storage)
cad_test(exchange)
cad_test(commands)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME bootstrap COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/bootstrap_test.py)

add_executable(storage-driver tests/storage_driver.cpp)
target_link_libraries(storage-driver PRIVATE cad)
add_test(NAME recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/recovery_test.py $<TARGET_FILE:storage-driver>)

add_executable(cad-runner src/runner.cpp)
target_link_libraries(cad-runner PRIVATE cad)
target_include_directories(cad-runner PRIVATE ${CAD_DEPS}/${JSON_DIR}/single_include)
add_test(NAME runner COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runner_test.py $<TARGET_FILE:cad-runner>)
