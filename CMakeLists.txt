cmake_minimum_required(VERSION 3.24)
project(FusionFoundation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
include(cmake/Dependencies.cmake)
include(cmake/PlaneGCS.cmake)
add_library(cad src/kernel.cpp src/sketch.cpp src/document.cpp src/selection.cpp)
target_include_directories(cad PUBLIC include PRIVATE src ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(cad PUBLIC ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(cad PUBLIC TKCAF TKLCAF TKCDF planegcs TKPrim TKBool TKTopAlgo TKBRep TKG3d TKMath TKernel)
function(cad_test name)
 add_executable(${name}_test tests/${name}_test.cpp)
 target_link_libraries(${name}_test PRIVATE cad)
 add_test(NAME ${name} COMMAND ${name}_test)
endfunction()
cad_test(kernel)
cad_test(sketch)
cad_test(document)
cad_test(selection)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME bootstrap COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/bootstrap_test.py)
